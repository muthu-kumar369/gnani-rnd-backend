## 11.3 Semantic Memory System

### Vector-Based Storage

Replace keyword matching with embeddings:

```typescript
class SemanticMemorySystem {
  async storeInteraction(interaction: Interaction): Promise<void> {
    // 1. Generate embeddings
    const userEmbedding = await this.embedder.embed(interaction.userMessage);
    const assistantEmbedding = await this.embedder.embed(interaction.assistantMessage);
    
    // 2. Extract metadata using LLM
    const metadata = await this.extractMetadata(interaction);
    
    // 3. Store in vector database
    await this.vectorStore.upsert({
      id: interaction.id,
      vectors: { user: userEmbedding, assistant: assistantEmbedding },
      metadata: {
        userId: interaction.userId,
        timestamp: interaction.timestamp,
        topics: metadata.topics,
        entities: metadata.entities,
        importance: metadata.importance // 0-1 score
      }
    });
  }
  
  async retrieveRelevant(query: string, userId: string, k: number = 10): Promise<Memory[]> {
    // 1. Embed query
    const queryEmbedding = await this.embedder.embed(query);
    
    // 2. Hybrid search
    const results = await this.vectorStore.search(queryEmbedding, {
      filter: { userId },
      limit: k * 2
    });
    
    // 3. Re-rank with hybrid scoring
    return this.hybridRerank(results, query);
  }
  
  private hybridRerank(results: VectorSearchResult[], query: string): Memory[] {
    return results.map(r => {
      const semanticScore = r.score; // 0-1
      const recencyScore = Math.exp(-(Date.now() - r.timestamp) / (7 * 24 * 60 * 60 * 1000));
      const importanceScore = r.metadata.importance || 0.5;
      const accessScore = Math.min(r.metadata.accessCount / 10, 1);
      
      r.finalScore = 
        0.5 * semanticScore +
        0.2 * recencyScore +
        0.2 * importanceScore +
        0.1 * accessScore;
      
      return r;
    }).sort((a, b) => b.finalScore - a.finalScore).slice(0, 10);
  }
}
```

class ToolChaining {
  async chainTools(query: string): Promise<ChainResult> {
    // Example: "What's the weather in the capital of France?"
    // Chain: get_capital(France) -> get_weather(Paris)
    
    // 1. Detect if chaining needed
    const needsChaining = await this.detectChaining(query);

      return this.executeSingleTool(query);
    }
    
    // 2. Plan the chain
    const chain = await this.planChain(query);
    
    // 3. Execute chain
    let result = null;
    for (const step of chain) {

      result = await this.executeTool(step.toolName, params);
    }
    
    return result;
  }
}

class IntelligentToolOrchestrator {
  async planToolExecution(query: string, tools: Tool[]): Promise<ExecutionPlan> {
    const prompt = `You are a tool planning expert. Create an execution plan.

User Query: "${query}"

Available Tools:
${tools.map(t => `- ${t.name}: ${t.description}`).join('\n')}

Create a step-by-step plan. For each step, specify:
1. What to do
2. Which tool to use (if any)
3. Parameters to pass
4. How to use the result

Return JSON with the plan.`;
    
    const plan = await this.llm.generate(prompt, { format: 'json', temperature: 0.1 });
    return this.parsePlan(plan);
  }
  
  async executePlan(plan: ExecutionPlan): Promise<ExecutionResult> {
    const results = [];
    let context = {};
    
    for (const step of plan.steps) {
      try {
        const result = await this.executeStep(step, context);
        results.push(result);
        context = { ...context, [step.id]: result.data };
        
        if (result.shouldStop) break;
      } catch (error) {
        const fallback = await this.handleStepFailure(step, error, context);
        if (fallback) {
          results.push(fallback);
          context = { ...context, [step.id]: fallback.data };
        } else {
          break;
        }
      }
    }
    
    return { steps: results, finalContext: context };
  }
}

class ConfidenceBasedResponse {
  async generateResponse(query: string, context: Context): Promise<string> {
    // 1. Generate initial response
    let response = await this.llm.generate(this.buildPrompt(query, context));
    
    // 2. Calculate confidence
    const confidence = await this.calculateConfidence(query, response, context);
    
    // 3. Adjust based on confidence
    if (confidence < 0.5) {
      // Low confidence - be honest
      response = await this.addUncertaintyQualifiers(response);
      response += "\n\nI'm not entirely certain. Would you like me to search for more information?";
    } else if (confidence < 0.7) {
      // Medium confidence - add caveats
      response = await this.addCaveats(response);
    }
    
    return response;
  }
  
  private async calculateConfidence(query: string, response: string, context: Context): Promise<number> {
    const factors = {
      contextRelevance: await this.measureContextRelevance(query, context),
      hasToolResult: context.toolResult ? 1.0 : 0.5,
      consistencyWithHistory: await this.checkConsistency(response, context.history),
      llmConfidence: await this.getLLMConfidence(response)
    };
    
    return (
      factors.contextRelevance * 0.3 +
      factors.hasToolResult * 0.3 +
      factors.consistencyWithHistory * 0.2 +
      factors.llmConfidence * 0.2
    );
  }
}

class StyleAdapter {
  async adaptStyle(response: string, userProfile: UserProfile): Promise<string> {
    // 1. Infer preferred style
    const style = await this.inferPreferredStyle(userProfile);
    
    // 2. Adjust response
    const adjustments = {
      technical: () => this.makeTechnical(response),
      casual: () => this.makeCasual(response),
      concise: () => this.makeConcise(response),
      detailed: () => this.makeDetailed(response)
    };
    
    return adjustments[style]?.() || response;
  }
  
  private async inferPreferredStyle(profile: UserProfile): Promise<Style> {
    const userMessages = profile.recentMessages.filter(m => m.role === 'user');
    
    const features = {
      avgLength: this.calculateAvgLength(userMessages),
      technicalTerms: this.countTechnicalTerms(userMessages),
      formalityScore: await this.analyzeFormalityScore(userMessages),
      emojiUsage: this.countEmojis(userMessages)
    };
    
    if (features.technicalTerms > 0.3 && features.formalityScore > 0.7) {
      return 'technical';
    } else if (features.emojiUsage > 0.1 && features.formalityScore < 0.4) {
      return 'casual';
    } else if (features.avgLength < 50) {
      return 'concise';
    }
    
    return 'balanced';
  }
}
